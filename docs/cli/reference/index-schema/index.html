    SQLite index schema¶

    index.sqlite holds everything in the output folder, normalized. Every method reference (defined in the APK or external) has one row in methods; edges point at integer ids, and the views call_edges, string_uses and search_items give the readable, descriptor-based form. JSON arrays are stored as TEXT; use SQLite's json_each to expand them. Query it with sextant sql <dir> "…" (read-only) or any SQLite tool.

    Schema version: 17 (stored in meta as index_schema_version).

    -- Key/value facts about the app and the index: package, version_name, version_code, min_sdk, target_sdk, label, sha256, index_schema_version, and meta_json (the full app metadata: signing, flags, icons, counts, warnings).
CREATE TABLE meta (
  key   TEXT PRIMARY KEY,
  value TEXT
);

-- Permissions the app requests (uses-permission, uses-permission-sdk-23), gets implicitly (implied) or defines (declared).
CREATE TABLE permissions (
  name             TEXT NOT NULL,
  kind             TEXT NOT NULL,   -- uses-permission | uses-permission-sdk-23 | implied | declared
  max_sdk          INTEGER,
  dangerous        INTEGER NOT NULL,
  protection_level TEXT,
  custom           INTEGER NOT NULL,
  implied_by       TEXT
);

-- Activities, activity aliases, services, receivers and providers from the manifest, with exported state, permissions, intent filters and review tags.
CREATE TABLE components (
  type            TEXT NOT NULL,    -- activity | activity-alias | service | receiver | provider
  name            TEXT NOT NULL,
  exported        INTEGER NOT NULL,
  exported_source TEXT NOT NULL,    -- explicit | default
  enabled         INTEGER NOT NULL,
  permission      TEXT,
  process         TEXT,
  launcher        INTEGER NOT NULL,
  target_activity TEXT,
  authorities     TEXT NOT NULL,    -- JSON array
  intent_filters  TEXT NOT NULL,    -- JSON array, same shape as components.jsonl
  read_permission         TEXT,     -- providers
  write_permission        TEXT,     -- providers
  grant_uri_permissions   INTEGER,  -- providers; NULL = not set
  path_permissions        TEXT NOT NULL, -- JSON array of <path-permission>
  launch_mode             TEXT,     -- activities
  task_affinity           TEXT,     -- activities
  foreground_service_type TEXT,     -- services
  isolated_process        INTEGER,
  direct_boot_aware       INTEGER,
  label                   TEXT,
  tags                    TEXT NOT NULL, -- JSON array of verdicts: launcher, app-link, deep-link, hidden-entry, task-hijack, …
  attributes              TEXT      -- JSON object of the other type-specific attributes that are set (launchMode, theme, …)
);

-- res/ and assets/ (index schema 7). String resources are separate from code strings (table strings).
CREATE TABLE string_resources (
  name    TEXT NOT NULL,
  kind    TEXT NOT NULL,            -- string | plurals | array
  config  TEXT NOT NULL,            -- '' default, fr, pt-rBR, night, …
  locale  TEXT,                     -- BCP-47, NULL for non-locale configs
  item    TEXT,                     -- plural quantity or array index
  value   TEXT NOT NULL
);
-- Every resource id in resources.arsc with its type, name and the configurations it exists in.
CREATE TABLE resource_ids (
  id      TEXT NOT NULL,            -- 0x7f0e001c
  type    TEXT NOT NULL,
  name    TEXT NOT NULL,
  configs TEXT NOT NULL             -- JSON array
);
-- Styles and themes with their parent and items (raw and resolved), and which components use them as a theme.
CREATE TABLE styles (
  name     TEXT NOT NULL,
  config   TEXT NOT NULL,
  parent   TEXT,
  items    TEXT NOT NULL,           -- JSON array of {name, raw, resolved}
  used_by  TEXT NOT NULL            -- JSON array: application / activity names using it as theme
);
-- Drawable and mipmap files: path inside the APK, format, size in bytes and pixel size.
CREATE TABLE drawables (
  type    TEXT NOT NULL,            -- drawable | mipmap
  name    TEXT NOT NULL,
  config  TEXT NOT NULL,
  path    TEXT NOT NULL,
  format  TEXT NOT NULL,            -- png | webp | vector | shape | selector | adaptive-icon | …
  bytes   INTEGER NOT NULL,
  width   INTEGER,
  height  INTEGER
);
-- Layout files: root view, number of views, ids, custom views, fragments and includes.
CREATE TABLE layouts (
  name         TEXT NOT NULL,
  config       TEXT NOT NULL,
  path         TEXT NOT NULL,
  root         TEXT,
  views        INTEGER NOT NULL,
  ids          TEXT NOT NULL,       -- JSON array
  custom_views TEXT NOT NULL,       -- JSON array
  fragments    TEXT NOT NULL,       -- JSON array
  includes     TEXT NOT NULL        -- JSON array
);
-- Menu items, one row per item or group, with title (raw and resolved), icon and showAsAction.
CREATE TABLE menus (
  menu           TEXT NOT NULL,
  config         TEXT NOT NULL,
  depth          INTEGER NOT NULL,
  tag            TEXT NOT NULL,     -- item | group | menu
  id             TEXT,
  title          TEXT,              -- as written (@string/x)
  title_resolved TEXT,
  icon           TEXT,
  show_as_action TEXT
);
-- Navigation graph destinations: kind, class, label, deep links and actions.
CREATE TABLE navigation_destinations (
  graph       TEXT NOT NULL,
  id          TEXT,
  kind        TEXT NOT NULL,        -- fragment | activity | dialog | navigation | …
  class_name  TEXT,
  label       TEXT,
  deep_links  TEXT NOT NULL,        -- JSON array
  actions     TEXT NOT NULL         -- JSON array of {id, destination, pop_up_to}
);
-- Files under assets/: kind, size, and whether the bytes are code (dex, ELF, zip) under another name.
CREATE TABLE assets (
  path        TEXT NOT NULL,
  kind        TEXT NOT NULL,
  bytes       INTEGER NOT NULL,
  magic       TEXT,                 -- dex | elf | zip when the content is code or an archive
  hidden_code INTEGER NOT NULL
);
-- res/xml configuration files (network security, backup rules, FileProvider paths, ...) with a summary and the decoded XML.
CREATE TABLE xml_configs (
  name    TEXT NOT NULL,
  config  TEXT NOT NULL,
  path    TEXT NOT NULL,
  kind    TEXT NOT NULL,            -- network-security-config | file-provider-paths | backup-rules | …
  summary TEXT NOT NULL,            -- JSON object
  xml     TEXT NOT NULL
);
-- Who refers to which resource: layouts, menus, navigation graphs, res/xml, styles, the manifest and code.
CREATE TABLE resource_refs (
  ref       TEXT NOT NULL,          -- @string/app_name
  from_kind TEXT NOT NULL,          -- layout | menu | navigation | xml | style | manifest | code (from = method, attribute = the first call after the id is loaded, usually the one it is passed to)
  "from"    TEXT NOT NULL,
  attribute TEXT NOT NULL
);

-- uses-feature / uses-library / uses-native-library
CREATE TABLE features (
  kind     TEXT NOT NULL,           -- feature | library | native-library
  name     TEXT NOT NULL,
  required INTEGER,                 -- NULL = not set (defaults to required)
  version  TEXT
);

-- meta-data and property on the application (scope = 'application') or a component (scope = class name)
CREATE TABLE metadata (
  scope    TEXT NOT NULL,
  kind     TEXT NOT NULL,           -- meta-data | property
  name     TEXT NOT NULL,
  value    TEXT,
  resource TEXT
);

-- <queries>: package visibility (Android 11+)
CREATE TABLE queries (
  kind       TEXT NOT NULL,         -- package | intent | provider
  name       TEXT NOT NULL,
  actions    TEXT NOT NULL,         -- JSON array
  categories TEXT NOT NULL,         -- JSON array
  data       TEXT NOT NULL          -- JSON array
);

-- URL patterns each component answers to, from VIEW intent filters.
CREATE TABLE deep_links (
  component TEXT NOT NULL,
  url       TEXT NOT NULL
);

-- Classes defined in the dex files of the APK.
CREATE TABLE classes (
  id           INTEGER PRIMARY KEY,
  descriptor   TEXT NOT NULL UNIQUE, -- Lcom/app/Foo;
  name         TEXT NOT NULL,        -- com.app.Foo
  package      TEXT NOT NULL,
  superclass   TEXT,
  interfaces   TEXT NOT NULL,        -- JSON array of descriptors
  access       TEXT NOT NULL,        -- space separated
  source_file  TEXT,
  dex          TEXT NOT NULL,
  kotlin       INTEGER NOT NULL,
  annotations  TEXT NOT NULL,        -- JSON array of descriptors
  method_count INTEGER NOT NULL,
  field_count  INTEGER NOT NULL
);

-- Every method reference. defined = 1 for methods with a definition in the APK; external ones have NULL details.
CREATE TABLE methods (
  id           INTEGER PRIMARY KEY,
  descriptor   TEXT NOT NULL UNIQUE, -- Lcom/app/Foo;->bar(I)V
  class        TEXT NOT NULL,        -- Lcom/app/Foo;
  name         TEXT NOT NULL,
  defined      INTEGER NOT NULL,
  params       TEXT,                 -- JSON array of type descriptors
  return_type  TEXT,
  access       TEXT,
  has_code     INTEGER,
  instructions INTEGER,
  registers    INTEGER,
  calls_out    INTEGER,
  string_loads INTEGER
);

-- Fields defined in the APK, with constant initial values.
CREATE TABLE fields (
  id            INTEGER PRIMARY KEY,
  descriptor    TEXT NOT NULL UNIQUE, -- Lcom/app/Foo;->count:I
  class         TEXT NOT NULL,
  name          TEXT NOT NULL,
  type          TEXT NOT NULL,
  access        TEXT NOT NULL,
  initial_value TEXT
);

-- Call edges between methods, by integer id; the call_edges view has readable names.
CREATE TABLE calls (
  caller  INTEGER NOT NULL REFERENCES methods(id),
  callee  INTEGER NOT NULL REFERENCES methods(id),
  kind    TEXT NOT NULL,             -- virtual | static | direct | super | interface | polymorphic
  count   INTEGER NOT NULL,
  offsets TEXT NOT NULL              -- JSON array, 16-bit code units, max 32
);

-- Distinct string constants loaded by code.
CREATE TABLE strings (
  id    INTEGER PRIMARY KEY,
  value TEXT NOT NULL UNIQUE
);

-- Which method loads which string constant, by integer id; the string_uses view has readable values.
CREATE TABLE string_refs (
  string  INTEGER NOT NULL REFERENCES strings(id),
  method  INTEGER NOT NULL REFERENCES methods(id),
  count   INTEGER NOT NULL,
  offsets TEXT NOT NULL
);

-- Field reads/writes and type uses (new, const-class, check-cast, instance-of, arrays) per method.
CREATE TABLE field_refs (
  method  INTEGER NOT NULL REFERENCES methods(id),
  field   TEXT NOT NULL,             -- Lcom/app/Foo;->count:I
  kind    TEXT NOT NULL,             -- read | write
  count   INTEGER NOT NULL,
  offsets TEXT NOT NULL
);

-- Type uses per method: new, const-class, check-cast, instance-of, arrays.
CREATE TABLE type_refs (
  method  INTEGER NOT NULL REFERENCES methods(id),
  type    TEXT NOT NULL,             -- Lcom/app/Foo;
  kind    TEXT NOT NULL,             -- new | class | cast | instanceof | array
  count   INTEGER NOT NULL,
  offsets TEXT NOT NULL
);

-- The calls table with readable method descriptors.
CREATE VIEW call_edges AS
  SELECT a.descriptor AS caller, b.descriptor AS callee, c.kind, c.count, c.offsets
  FROM calls c JOIN methods a ON a.id = c.caller JOIN methods b ON b.id = c.callee;

-- The string_refs table with the string value and the method descriptor.
CREATE VIEW string_uses AS
  SELECT s.value, m.descriptor AS method, r.count, r.offsets
  FROM string_refs r JOIN strings s ON s.id = r.string JOIN methods m ON m.id = r.method;

-- Version 5 insights.
CREATE TABLE libraries (
  "group"  TEXT NOT NULL,
  artifact TEXT NOT NULL,
  version  TEXT NOT NULL,
  evidence TEXT NOT NULL
);

-- Cross-platform frameworks detected (Flutter, React Native, .NET MAUI, Unity, ...), with evidence.
CREATE TABLE frameworks (
  name     TEXT NOT NULL,
  evidence TEXT NOT NULL             -- JSON array
);

-- Known SDKs by category (analytics, ads, crash reporting, ...), with class counts, packages and manifest evidence.
CREATE TABLE sdks (
  name     TEXT NOT NULL,
  category TEXT NOT NULL,
  classes  INTEGER NOT NULL,
  packages TEXT NOT NULL,            -- JSON array
  manifest TEXT NOT NULL             -- JSON array
);

-- Every distinct URL found in the APK.
CREATE TABLE endpoints (
  id     INTEGER PRIMARY KEY,
  url    TEXT NOT NULL UNIQUE,
  scheme TEXT NOT NULL,
  host   TEXT NOT NULL
);

-- Where each URL was found: a method, a resource, a deep link, the manifest, an asset or a native library.
CREATE TABLE endpoint_sources (
  endpoint INTEGER NOT NULL REFERENCES endpoints(id),
  kind     TEXT NOT NULL,            -- code | resource | deep_link | manifest | asset | native
  location TEXT NOT NULL             -- method descriptor, @string/name or component
);

-- Default values of simple resources (string, bool, integer, color, dimen); every configuration is in resource_values.
CREATE TABLE resources (
  type  TEXT NOT NULL,               -- string | bool | integer | color | dimen
  name  TEXT NOT NULL,
  value TEXT NOT NULL
);

-- Native .so files per ABI: size, sha256, ELF class, exported symbols and JNI methods.
CREATE TABLE native_libraries (
  path             TEXT NOT NULL,
  abi              TEXT NOT NULL,
  name             TEXT NOT NULL,
  size_bytes       INTEGER NOT NULL,
  sha256           TEXT NOT NULL,
  elf_class        TEXT,
  exported_symbols INTEGER NOT NULL,
  jni_methods      INTEGER NOT NULL
);

-- Exported symbols of native libraries; JNI symbols are mapped back to the Java class and method they implement.
CREATE TABLE native_symbols (
  library    TEXT NOT NULL,
  name       TEXT NOT NULL,
  jni_class  TEXT,
  jni_method TEXT
);

-- Every zip entry of the APK (index schema 8): the bottom layer, so every byte is accounted for.
-- decoded_by names the decoder that turned the file into rows; NULL = listed only. See `sextant coverage`.
CREATE TABLE files (
  id               INTEGER PRIMARY KEY,
  path             TEXT NOT NULL UNIQUE,
  type             TEXT NOT NULL,  -- manifest | resource-table | dex | resource | native-library | asset | signature | library-version | kotlin-metadata | service-loader | meta-inf | other
  size_bytes       INTEGER NOT NULL,
  compressed_bytes INTEGER NOT NULL,
  method           TEXT NOT NULL,  -- stored | deflated
  crc32            TEXT NOT NULL,  -- hex, from the zip header
  sha256           TEXT,           -- of the uncompressed bytes; NULL = unreadable or over 1 GB
  magic            TEXT,           -- content from the first bytes: dex | elf | zip | png | webp | jpeg | gif | binary-xml | resource-table | xml | json | font | text
  resource         TEXT,           -- @layout/main: the resource this res/ file is (shrunk apps rename files)
  config           TEXT,           -- that resource's configuration: '' default, xxxhdpi-v4, night, …
  decoded_by       TEXT            -- manifest | resource-table | code | layouts | menus | navigation | xml-configs | drawables | assets | native | libraries | signing
);

-- The raw layer (index schema 10): every binary XML file of the APK (manifest, layouts, menus, drawables, colors,
-- animations, …) node by node, so any view or attribute can be queried. `sextant xml` prints a file rebuilt from these rows.
CREATE TABLE xml_nodes (
  id     INTEGER PRIMARY KEY,
  file   INTEGER,                 -- files.id
  path   TEXT NOT NULL,           -- the file inside the APK, e.g. res/BW.xml
  parent INTEGER,                 -- xml_nodes.id; NULL for the root element
  depth  INTEGER NOT NULL,        -- 0 for the root
  seq    INTEGER NOT NULL,        -- document order within the file
  tag    TEXT NOT NULL,           -- LinearLayout, com.app.ui.Card, item, vector, …
  text   TEXT                     -- element text, when it has no child elements
);
-- Attributes of xml_nodes, as decoded, with ref set when the value points at a resource.
CREATE TABLE xml_attrs (
  node  INTEGER NOT NULL,         -- xml_nodes.id
  name  TEXT NOT NULL,            -- android:text, app:layout_constraintTop_toTopOf, …
  value TEXT NOT NULL,            -- as decoded: @string/title, 16.0dip, #ff000000, match_parent
  ref   TEXT                      -- the resource it points at (@string/title, ?attr/colorPrimary, @android:color/white); NULL for plain values
);

-- resources.arsc in every configuration (index schema 10). Simple values and file paths in resource_values;
-- bags (styles, plurals, arrays, attr enums and flags) item by item in resource_bag_items.
CREATE TABLE resource_values (
  id     TEXT NOT NULL,           -- 0x7f0e001c
  type   TEXT NOT NULL,
  name   TEXT NOT NULL,
  config TEXT NOT NULL,           -- '' default, fr, night, xxhdpi-v4, …
  value  TEXT,                    -- the value, or the file path for file resources; NULL for bags
  parent TEXT                     -- bags: parent style
);
-- Items of bag resources: style attributes, plural quantities, array items, attr enum and flag values.
CREATE TABLE resource_bag_items (
  id     TEXT NOT NULL,           -- resource_values.id
  config TEXT NOT NULL,
  seq    INTEGER NOT NULL,
  key    TEXT NOT NULL,           -- style attribute, plural quantity, array index or enum/flag name
  value  TEXT NOT NULL
);

-- Links (index schema 11): every edge between kinds of things, with readable keys, built from the tables above.
-- Keys: app, component:<name>, class:<descriptor>, method:<descriptor>, permission:<name>, @type/name,
-- url:<url>, native:<path>, file:<path>. rel: declares | requests | defines | implemented_by | aliases | requires |
-- opens | uses | instantiates | contains | implements. `sextant links` walks them, adding calls on the fly.
CREATE TABLE links (
  src TEXT NOT NULL,
  rel TEXT NOT NULL,
  dst TEXT NOT NULL,
  via TEXT                        -- evidence: the attribute, the call it is passed to, the component type, …
);

-- Every key that appears in links, with its kind (the part before ':', or resource for @type/name).
CREATE VIEW entities AS
  SELECT key, CASE WHEN key = 'app' THEN 'app' WHEN key LIKE '@%' OR key LIKE '?%' THEN 'resource'
                   ELSE substr(key, 1, instr(key, ':') - 1) END AS kind
  FROM (SELECT src AS key FROM links UNION SELECT dst FROM links);

-- What every table, view and column means (index schema 12), from the comments in this schema. Start here:
-- SELECT * FROM schema_docs WHERE "table" = 'links'
CREATE TABLE schema_docs (
  "table"     TEXT NOT NULL,
  "column"    TEXT,               -- NULL for the row that describes the table or view itself
  type        TEXT NOT NULL,      -- table | view | virtual table, or the column's SQL type
  description TEXT
);

-- Full records (index schema 9): every manifest, resource and insight record exactly as Sextant built it, one JSON
-- row each, so a command can rebuild the whole extraction from this file alone. The tables above are the
-- queryable view of the same facts. kind: manifest | permission | component | feature | metadata | query | resources | insights
CREATE TABLE records (
  kind TEXT NOT NULL,
  seq  INTEGER NOT NULL,
  json TEXT NOT NULL
);

-- Every ELF file in the APK, read in depth (index schema 14): native libraries under lib/ and ELF bytes anywhere else.
CREATE TABLE elf_files (
  path             TEXT PRIMARY KEY,   -- the file inside the APK
  file             INTEGER,            -- files.id
  elf_class        TEXT NOT NULL,      -- ELF32 | ELF64
  machine          TEXT NOT NULL,      -- arm64 | arm | x86 | x86_64 | riscv64 | …
  soname           TEXT,               -- DT_SONAME
  build_id         TEXT,               -- GNU build id, hex
  runpath          TEXT,               -- DT_RUNPATH or DT_RPATH
  exported         INTEGER NOT NULL,   -- defined dynamic symbols
  imported         INTEGER NOT NULL,   -- undefined dynamic symbols (from other libraries)
  strings          INTEGER NOT NULL,   -- rows in elf_strings
  strings_capped   INTEGER NOT NULL    -- 1 when elf_strings stopped at 20,000 for this file
);
-- Section headers of each ELF file.
CREATE TABLE elf_sections (
  path   TEXT NOT NULL,
  seq    INTEGER NOT NULL,
  name   TEXT NOT NULL,                -- .text, .rodata, .dynsym, …
  type   TEXT NOT NULL,                -- progbits | nobits | dynsym | dynamic | note | …
  offset INTEGER NOT NULL,
  size   INTEGER NOT NULL,
  flags  TEXT NOT NULL                 -- w (write), a (alloc), x (exec)
);
-- The libraries each ELF file loads (DT_NEEDED), in order.
CREATE TABLE elf_needed (
  path     TEXT NOT NULL,
  seq      INTEGER NOT NULL,
  needed   TEXT NOT NULL,              -- libc.so, libc++_shared.so, …
  resolved TEXT                        -- the library in the APK it resolves to (same folder); NULL = from the system
);
-- Every dynamic symbol: exported (defined = 1) or imported from another library (defined = 0).
CREATE TABLE elf_symbols (
  path    TEXT NOT NULL,
  name    TEXT NOT NULL,
  type    TEXT NOT NULL,               -- func | object | other
  binding TEXT NOT NULL,               -- global | weak | local
  defined INTEGER NOT NULL
);
-- Printable strings (6+ characters) in the read-only data of each ELF file, at most 20,000 per file.
CREATE TABLE elf_strings (
  path   TEXT NOT NULL,
  offset INTEGER NOT NULL,             -- file offset
  value  TEXT NOT NULL
);

-- Text files (index schema 15): the text of every file whose content is text, JSON or textual XML, up to 4 MB
-- (assets, library metadata, licenses, configs, res/raw). JSON works with json_extract(text, '$.key').
CREATE TABLE file_text (
  path  TEXT PRIMARY KEY,         -- the file inside the APK
  file  INTEGER,                  -- files.id
  lines INTEGER NOT NULL,
  text  TEXT NOT NULL
);
-- Java .properties files, key by key in file order.
CREATE TABLE file_properties (
  path  TEXT NOT NULL,
  seq   INTEGER NOT NULL,
  key   TEXT NOT NULL,
  value TEXT NOT NULL
);
-- Service-loader registrations (META-INF/services/<interface>): which class implements which interface.
CREATE TABLE services (
  interface      TEXT NOT NULL,   -- com.example.Plugin
  implementation TEXT NOT NULL,   -- com.example.impl.PluginImpl
  path           TEXT NOT NULL    -- the META-INF/services file
);
-- Full-text search over file_text (any 3+ character fragment); `sextant files --grep` uses it.
CREATE VIRTUAL TABLE file_text_search USING fts5(text, content = 'file_text', tokenize = 'trigram');

-- Code details (index schema 17): every annotation on a class, method, field or parameter, with its values.
CREATE TABLE annotations (
  id          INTEGER PRIMARY KEY,
  target_kind TEXT NOT NULL,      -- class | method | field | parameter
  target      TEXT NOT NULL,      -- the class, method or field descriptor
  parameter   INTEGER,            -- parameter annotations: the parameter index
  type        TEXT NOT NULL,      -- Lkotlin/Metadata;, Ljavax/inject/Inject;, …
  visibility  TEXT NOT NULL       -- build | runtime | system
);
-- Element values of annotations: strings and numbers plain, arrays and nested values in smali syntax (at most 4,000 characters).
CREATE TABLE annotation_values (
  annotation INTEGER NOT NULL,    -- annotations.id
  name       TEXT NOT NULL,
  value      TEXT NOT NULL
);
-- Source lines of each method from its debug info: the range, and every code offset with its line.
CREATE TABLE method_lines (
  method     INTEGER NOT NULL REFERENCES methods(id),
  first_line INTEGER NOT NULL,
  last_line  INTEGER NOT NULL,
  line_table TEXT NOT NULL        -- "offset:line offset:line …", offsets in 16-bit code units
);
-- Parameter names from debug info (missing where the build stripped them).
CREATE TABLE method_params (
  method INTEGER NOT NULL REFERENCES methods(id),
  idx    INTEGER NOT NULL,
  name   TEXT NOT NULL
);
-- Exception handlers: code in [start, end) jumps to handler when it throws type (NULL = any, as for finally).
CREATE TABLE exception_handlers (
  method  INTEGER NOT NULL REFERENCES methods(id),
  start   INTEGER NOT NULL,
  "end"   INTEGER NOT NULL,
  handler INTEGER NOT NULL,
  type    TEXT
);
-- Kotlin classes from their @kotlin.Metadata: kind k (1 class, 2 file facade, 3 synthetic, 4 multi-file facade, 5 multi-file part) and metadata version.
CREATE VIEW kotlin_classes AS
  SELECT a.target AS class,
         (SELECT value FROM annotation_values v WHERE v.annotation = a.id AND v.name = 'k') AS kind,
         (SELECT value FROM annotation_values v WHERE v.annotation = a.id AND v.name = 'mv') AS version
  FROM annotations a WHERE a.type = 'Lkotlin/Metadata' || char(59) AND a.target_kind = 'class';  -- char(59) is ';', which the statement splitter would cut on

-- Signing as rows (index schema 16): every signer of every scheme as apksig reports it.
CREATE TABLE signing_signers (
  scheme      TEXT NOT NULL,      -- v1 | v2 | v3 | v3.1 | v4 | source-stamp
  idx         INTEGER NOT NULL,   -- signer index within the scheme
  name        TEXT,               -- v1: the signature file, e.g. META-INF/CERT.SF
  min_sdk     INTEGER,            -- v3 / v3.1: lowest SDK this signer applies to
  max_sdk     INTEGER,            -- v3 / v3.1: highest SDK this signer applies to
  certificate TEXT                -- signing_certificates.sha256
);
-- Content digests each v2 / v3 signer signed.
CREATE TABLE signing_digests (
  scheme    TEXT NOT NULL,
  idx       INTEGER NOT NULL,
  algorithm TEXT NOT NULL,        -- RSA_PKCS1_V1_5_WITH_SHA256, ECDSA_WITH_SHA256, …
  digest    TEXT NOT NULL         -- hex
);
-- Every certificate seen (signers, rotation lineage, source stamp), with every field people check and its DER bytes.
CREATE TABLE signing_certificates (
  sha256              TEXT PRIMARY KEY,   -- colon-separated hex, as in meta_json
  sha1                TEXT NOT NULL,
  md5                 TEXT NOT NULL,
  subject             TEXT NOT NULL,
  issuer              TEXT NOT NULL,
  serial              TEXT NOT NULL,      -- hex
  not_before          TEXT NOT NULL,
  not_after           TEXT NOT NULL,
  signature_algorithm TEXT NOT NULL,      -- SHA256withRSA, …
  key_algorithm       TEXT NOT NULL,      -- RSA | EC | DSA
  key_size            INTEGER,            -- bits
  public_key_sha256   TEXT NOT NULL,
  der                 BLOB NOT NULL
);
-- Key rotation (v3 proof-of-rotation): the signing certificates in order, oldest first.
CREATE TABLE signing_lineage (
  seq         INTEGER NOT NULL,
  certificate TEXT NOT NULL       -- signing_certificates.sha256
);
-- What the verifier reported: errors and warnings for the APK, each signer and the source stamp.
CREATE TABLE signing_issues (
  level   TEXT NOT NULL,          -- error | warning
  scope   TEXT NOT NULL,          -- apk | v1:<name> | v2:<index> | v3:<index> | v3.1:<index> | source-stamp
  issue   TEXT NOT NULL,          -- apksig issue id, e.g. JAR_SIG_NO_SIGNATURES
  message TEXT NOT NULL
);
-- Every digest in META-INF/MANIFEST.MF (v1): one row per entry and algorithm.
CREATE TABLE signing_manifest (
  path      TEXT NOT NULL,        -- the entry it covers
  algorithm TEXT NOT NULL,        -- SHA-256, SHA1, …
  digest    TEXT NOT NULL         -- base64, as written
);

-- File contents (index schema 13): the uncompressed bytes of every file in the APK (dex files always; the others
-- unless extracted with --no-content), so the .sqlite file holds the whole APK and no command needs the APK itself.
CREATE TABLE file_contents (
  file  INTEGER PRIMARY KEY,      -- files.id
  bytes BLOB NOT NULL
);

-- Launcher icon files: role is icon | round_icon, or <role>.background/.foreground/.monochrome for adaptive layers.
CREATE TABLE icons (
  role       TEXT NOT NULL,
  resource   TEXT NOT NULL,          -- @mipmap/ic_launcher
  path       TEXT NOT NULL,          -- file inside the APK
  config     TEXT NOT NULL,          -- xxxhdpi-v4, anydpi-v26, …
  density    INTEGER,                -- dpi; NULL for anydpi/nodpi
  format     TEXT NOT NULL,          -- png | webp | jpg | gif | xml
  width      INTEGER,
  height     INTEGER,
  size_bytes INTEGER NOT NULL,
  best       INTEGER NOT NULL        -- 1 for the file Sextant picks to show
);

-- Review findings (static observations to check, not proven vulnerabilities).
CREATE TABLE findings (
  id       TEXT NOT NULL,
  severity TEXT NOT NULL,            -- high | medium | low | info
  title    TEXT NOT NULL,
  detail   TEXT NOT NULL,
  count    INTEGER NOT NULL
);

-- The evidence behind each finding: a manifest entry, method, class, string or resource.
CREATE TABLE finding_evidence (
  finding  TEXT NOT NULL,            -- findings.id
  kind     TEXT NOT NULL,            -- manifest | method | class | string | resource
  location TEXT NOT NULL,
  value    TEXT
);

-- Substring search (trigram: any 3+ character fragment matches). Contentless to save space:
-- rowid = (source id << 4) | kind; kinds: 1 class, 2 method, 3 field, 4 string, 5 component, 6 permission,
-- 7 deep_link, 8 resource, 9 endpoint, 10 sdk, 11 library, 12 native_symbol, 13 finding, 14 string_resource, 15 native_string.
-- Indexed text: class full name; method and field "SimpleClass.name" (field + " = value"); first 1000 chars of strings.
CREATE VIRTUAL TABLE search USING fts5(text, content = '', tokenize = 'trigram');

-- What each search rowid points at, for SQL users:
--   SELECT i.kind, i.ref, i.text FROM search JOIN search_items i ON i.search_rowid = search.rowid WHERE search MATCH 'Cipher'
CREATE VIEW search_items AS
  SELECT id * 16 + 1 AS search_rowid, 'class' AS kind, descriptor AS ref, name AS text FROM classes
  UNION ALL SELECT id * 16 + 2, 'method', descriptor, descriptor FROM methods
  UNION ALL SELECT id * 16 + 3, 'field', descriptor, descriptor || coalesce(' = ' || initial_value, '') FROM fields
  UNION ALL SELECT id * 16 + 4, 'string', id, value FROM strings
  UNION ALL SELECT rowid * 16 + 5, 'component', name, type || ' ' || name FROM components
  UNION ALL SELECT rowid * 16 + 6, 'permission', name, name FROM permissions
  UNION ALL SELECT rowid * 16 + 7, 'deep_link', component, url FROM deep_links
  UNION ALL SELECT rowid * 16 + 8, 'resource', '@' || type || '/' || name, name || ' = ' || value FROM resources
  UNION ALL SELECT id * 16 + 9, 'endpoint', url, url FROM endpoints
  UNION ALL SELECT rowid * 16 + 10, 'sdk', name, name || ' (' || category || ')' FROM sdks
  UNION ALL SELECT rowid * 16 + 11, 'library', "group" || ':' || artifact, "group" || ':' || artifact || ':' || version FROM libraries
  UNION ALL SELECT rowid * 16 + 12, 'native_symbol', library, name FROM native_symbols
  UNION ALL SELECT rowid * 16 + 13, 'finding', id, severity || ' ' || title FROM findings
  UNION ALL SELECT rowid * 16 + 14, 'string_resource', '@string/' || name, name || ' = ' || value FROM string_resources
  UNION ALL SELECT min(rowid) * 16 + 15, 'native_string', path, value FROM elf_strings GROUP BY value;
